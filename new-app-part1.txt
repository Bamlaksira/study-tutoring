function track(eventName, data = {}) {
  try {
    if (typeof window.fbq === 'function') {
      window.fbq('trackCustom', eventName, data)
    }
  } catch {}
}

function getTrackingData() {
  const params = new URLSearchParams(window.location.search)

  return {
    utmSource: params.get('utm_source') || '',
    utmMedium: params.get('utm_medium') || '',
    utmCampaign: params.get('utm_campaign') || '',
    utmContent: params.get('utm_content') || '',
    utmTerm: params.get('utm_term') || '',
    landingPage: window.location.href,
    referrer: document.referrer || ''
  }
}

function App() {
  if (window.location.pathname === '/onboarding') {
    return <Onboarding />
  }

  const [language, setLanguage] = useState(
    () => localStorage.getItem('studycareLanguage') || 'en'
  )
  const [menuOpen, setMenuOpen] = useState(false)
  const [showGuideForm, setShowGuideForm] = useState(false)
  const [showPrivacy, setShowPrivacy] = useState(
    window.location.pathname === '/privacy-policy'
  )
  const [guideSuccess, setGuideSuccess] = useState(false)
  const [formError, setFormError] = useState('')
  const [submitting, setSubmitting] = useState(false)
  const [assessmentStarted, setAssessmentStarted] = useState(false)
  const [assessmentAnswers, setAssessmentAnswers] = useState({})
  const [assessmentComplete, setAssessmentComplete] = useState(false)
  const [openFaq, setOpenFaq] = useState(null)

  const guideFormRef = useRef(null)

  const t = translations[language]

  useEffect(() => {
    localStorage.setItem('studycareLanguage', language)
  }, [language])

  useEffect(() => {
    const handlePopState = () => {
      setShowPrivacy(window.location.pathname === '/privacy-policy')
    }

    window.addEventListener('popstate', handlePopState)

    return () => window.removeEventListener('popstate', handlePopState)
  }, [])

  function goTo(path) {
    window.history.pushState({}, '', path)
    setShowPrivacy(path === '/privacy-policy')
    setMenuOpen(false)
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  function scrollToSection(id) {
    setMenuOpen(false)

    if (window.location.pathname !== '/') {
      window.history.pushState({}, '', '/')
      setShowPrivacy(false)
      setTimeout(() => {
        document.getElementById(id)?.scrollIntoView({ behavior: 'smooth' })
      }, 50)
      return
    }

    document.getElementById(id)?.scrollIntoView({ behavior: 'smooth' })
  }

  function openGuide() {
    track('StudyGuideFormStarted')
    setShowGuideForm(true)
    setGuideSuccess(false)
    setFormError('')
  }

  function closeGuide() {
    if (!submitting) {
      setShowGuideForm(false)
      setFormError('')
    }
  }

  function startAssessment() {
    setAssessmentStarted(true)
    setAssessmentComplete(false)
    track('ParentAssessmentStarted')
    setTimeout(() => {
      document.getElementById('assessment')?.scrollIntoView({
        behavior: 'smooth'
      })
    }, 50)
  }

  function answerAssessment(index, answer) {
    setAssessmentAnswers(prev => ({
      ...prev,
      [index]: answer
    }))
  }

  function completeAssessment() {
    if (Object.keys(assessmentAnswers).length < assessmentQuestions.length) {
      return
    }

    setAssessmentComplete(true)
    track('ParentAssessmentCompleted', {
      answeredQuestions: assessmentQuestions.length
    })
  }

  function getAssessmentResult() {
    const scores = {
      Often: 2,
      Sometimes: 1,
      Rarely: 0
    }

    const score = Object.values(assessmentAnswers).reduce(
      (total, answer) => total + (scores[answer] || 0),
      0
    )

    if (score >= 8) {
      return {
        title: language === 'am'
          ? 'የተዋቀረ ተጨማሪ ድጋፍ ሊጠቅም ይችላል'
          : 'Your child may benefit from structured extra support.',
        text: language === 'am'
          ? 'ይህ የምርመራ ውጤት ህክምናዊ ወይም የሙያ ምርመራ አይደለም። የጥናት ልምድን ለመገንዘብ የሚረዳ አጠቃላይ መመሪያ ብቻ ነው።'
          : 'This is not a diagnosis or professional assessment. It is only a general educational reflection on your child’s study experience.'
      }
    }

    if (score >= 4) {
      return {
        title: language === 'am'
          ? 'አንዳንድ የጥናት ድጋፍ ሊጠቅም ይችላል'
          : 'Some additional study support may be useful.',
        text: language === 'am'
          ? 'ይህ ውጤት የልጅዎን የጥናት ልምድ ለመመልከት የሚረዳ አጠቃላይ መመሪያ ነው።'
          : 'This is a general educational reflection, not a diagnosis or professional assessment.'
      }
    }

    return {
      title: language === 'am'
        ? 'የአሁኑን የጥናት ልምድ መቀጠል ይችላሉ'
        : 'Your child may be managing their current study experience reasonably well.',
      text: language === 'am'
        ? 'እንደ ፍላጎት የጥናት ልምዶችን ማሻሻል እና መከታተል ጠቃሚ ነው።'
        : 'It can still be useful to continue observing study habits and support needs over time.'
    }
  }

  async function handleGuideSubmit(event) {
    event.preventDefault()
    setFormError('')

    const formData = new FormData(event.currentTarget)

    const parentName = formData.get('parentName')?.trim()
    const phone = formData.get('phone')?.trim()
    const grade = formData.get('grade')
    const city = formData.get('city')?.trim()
    const country = formData.get('country')?.trim()

    if (!parentName || !phone || !grade || !city || !country) {
      setFormError(
        language === 'am'
          ? 'እባክዎ የሚያስፈልጉትን መስኮች በሙሉ ይሙሉ።'
          : 'Please complete all required fields.'
      )
      return
    }

    setSubmitting(true)

    const tracking = getTrackingData()

    const payload = {
      parentName,
      phone,
      grade,
      city,
      location: city,
      area: formData.get('area')?.trim() || '',
      country,
      preferredLanguage: formData.get('preferredLanguage') || language,
      mainLearningChallenge:
        formData.get('mainLearningChallenge')?.trim() || '',
      marketingSource: formData.get('marketingSource') || '',
      heardAbout: formData.get('marketingSource') || '',
      ...tracking
    }

    try {
      const response = await fetch(`${API}/api/leads`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(payload)
      })

      const data = await response.json().catch(() => ({}))

      if (!response.ok || !data.success) {
        throw new Error(data.message || 'Unable to submit form')
      }

      track('StudyGuideSubmitted', {
        grade,
        preferredLanguage: payload.preferredLanguage,
        marketingSource: payload.marketingSource
      })

      setGuideSuccess(true)
      event.currentTarget.reset()
    } catch (error) {
      setFormError(
        language === 'am'
          ? 'መረጃዎን ማስገባት አልተቻለም። እባክዎ እንደገና ይሞክሩ።'
          : 'We could not submit your information. Please try again.'
      )
    } finally {
      setSubmitting(false)
    }
  }

  function handleGuideDownload() {
    track('StudyGuideDownloaded')
  }

  function handleWhatsApp(message) {
    track('WhatsAppClicked')
    window.open(makeWhatsAppLink(message), '_blank', 'noopener,noreferrer')
  }

  function handlePaidClick() {
    track('PaidServiceClicked')
    goTo('/onboarding')
  }

  if (showPrivacy) {
    return (
      <div className="privacy-page">
        <div className="container privacy-content">
          <button className="back-button" onClick={() => goTo('/')}>
            ← {t.navHome}
          </button>

          <h1>Privacy Policy</h1>
          <p>Last updated: August 31, 2026</p>

          <h2>Information We Collect</h2>
          <p>
            When you use StudyCare forms, we may collect information such as
            your name, phone number, child’s grade, location, preferred
            language, learning challenge, and how you heard about StudyCare.
          </p>

          <h2>How We Use Information</h2>
          <p>
            We use submitted information to respond to inquiries, provide
            requested resources, understand learning support needs, and
            communicate with parents about StudyCare services.
          </p>

          <h2>Contact</h2>
          <p>
            If you have questions about information submitted through the
            website, please contact StudyCare.
          </p>
        </div>
      </div>
    )
  }

  return (
    <>
      <header className="navbar">
        <div className="container nav-inner">
          <button className="logo" onClick={() => scrollToSection('home')}>
            StudyCare
          </button>

          <button
            className="mobile-menu-button"
            onClick={() => setMenuOpen(!menuOpen)}
            aria-label="Toggle navigation"
          >
            ☰
          </button>

          <nav className={menuOpen ? 'nav-links open' : 'nav-links'}>
            <button onClick={() => scrollToSection('home')}>
              {t.navHome}
            </button>
            <button onClick={() => scrollToSection('services')}>
              {t.navServices}
            </button>
            <button onClick={() => scrollToSection('how-it-works')}>
              {t.navHow}
            </button>
            <button onClick={() => scrollToSection('resources')}>

                    {guideSubmitting ? t.submitting : t.submitGuide}
                  </button>

                  <p className="form-note">{t.formNote}</p>
                </form>
              </>
            ) : (
              <div className="guide-success">
                <div className="success-icon">✓</div>
                <span className="eyebrow">{t.successEyebrow}</span>
                <h2>{t.successTitle}</h2>
                <p>{t.successText}</p>

                <a
                  className="btn btn-primary"
                  href={GUIDE}
                  target="_blank"
                  rel="noopener noreferrer"
                  onClick={handleGuideDownload}
                >
                  {t.downloadGuide}
                </a>

                <div className="success-divider">
                  <span>{t.or}</span>
                </div>

                <p className="success-next">{t.successNext}</p>

                <button
                  className="btn btn-secondary"
                  onClick={() => handleWhatsApp(t.whatsappGeneral)}
                >
                  {t.talk}
                </button>
              </div>
            )}
          </div>
        </div>
      )}
    </>
  )
}

export default App
